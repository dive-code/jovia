    ========================== */

    function safeUrl(value) {

      try {

        const url = new URL(
          String(value || "").trim()
        );

        if (
          url.protocol !== "http:" &&
          url.protocol !== "https:"
        ) {
          return "#";
        }

        return url.href;

      } catch {

        return "#";

      }

    }

    function formatNaira(value) {

      const amount = Number(value || 0);

      return "₦" +
        amount.toLocaleString("en-NG");

    }

    /* =========================
       FORMAT DATE
    ========================== */

    function formatDate(value) {

      if (!value) {
        return "-";
      }

      const raw = String(value);

      let date;

      if (
        raw.includes(" ") &&
        !raw.includes("T")
      ) {

        date = new Date(
          raw.replace(" ", "T") + "Z"
        );

      } else {

        date = new Date(raw);

      }

      if (Number.isNaN(date.getTime())) {
        return raw;
      }

      return date.toLocaleString(
        "en-NG",
        {
          dateStyle: "medium",
          timeStyle: "short"
        }
      );

    }

              })

          data = {};

          );

          loginError.message ||
          "Unable to login.";

        error.style.display = "block";

        button.disabled = false;
        button.textContent = "Login to Admin";
        loading.style.display = "none";

    async function resetPassword(id) {

      const user =
        usersCache.get(Number(id));

      if (!user) {

        alert(
          "User information could not be found. Please refresh the users list."
        );

        return;

      }

      const username =
        user.username || "this user";

      const newPassword =
        prompt(
          `Enter a new password for ${username}:\n\nMinimum 8 characters.`
        );

      if (newPassword === null) {
        return;
      }

      if (newPassword.length < 8) {

        alert(
          "Password must contain at least 8 characters."
        );

        return;

      }

      const confirmPassword =
        prompt(
          "Enter the new password again to confirm:"
        );

      if (confirmPassword === null) {
        return;
      }

      if (newPassword !== confirmPassword) {

        alert(
          "The passwords do not match."
        );

        return;

      }

      try {

        const response =
          await fetch(
            `/api/admin/users/${Number(id)}/reset-password`,
            {
              method: "POST",

              headers: {
                "Content-Type":
                  "application/json"
              },

              credentials: "same-origin",

              body: JSON.stringify({
                newPassword: newPassword
              })

            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        let data;

        try {
          data = await response.json();
        } catch {
          data = {};
        }

        if (
          !response.ok ||
          !data.success
        ) {

          throw new Error(
            data.message ||
            "Unable to reset password."
          );

        }

        alert(
          data.message ||
          "Password reset successfully."
        );

      } catch (error) {

        console.error(
          "Password reset error:",
          error
        );

        alert(
          error.message ||
          "Unable to connect to the server."
        );

      }

    }

    async function createJob(event) {

      event.preventDefault();

      const slot =
        Number(
          document.getElementById(
            "jobSlot"
          ).value
        );

      const title =
        document.getElementById(
          "jobTitle"
        ).value.trim();

      const category =
        document.getElementById(
          "jobCategory"
        ).value.trim();

      const url =
        document.getElementById(
          "jobUrl"
        ).value.trim();

      const description =
        document.getElementById(
          "jobDescription"
        ).value.trim();

      const reward =
        Number(
          document.getElementById(
            "jobReward"
          ).value
        );

      const duration =
        document.getElementById(
          "jobDuration"
        ).value.trim();

      const status =
        document.getElementById(
          "jobStatus"
        ).value;

      if (![1, 2, 3].includes(slot)) {

        alert(
          "Job slot must be Job 1, Job 2 or Job 3."
        );

        return;

      }

      if (!title) {

        alert(
          "Please enter a job title."
        );

        return;

      }

      if (!category) {

        alert(
          "Please enter a job category."
        );

        return;

      }

      try {

        const parsedUrl =
          new URL(url);

        if (
          parsedUrl.protocol !== "http:" &&
          parsedUrl.protocol !== "https:"
        ) {

          throw new Error();

        }

      } catch {

        alert(
          "Please enter a valid HTTP or HTTPS job URL."
        );

        return;

      }

      if (!description) {

        alert(
          "Please enter a job description."
        );

        return;

      }

      if (
        !Number.isFinite(reward) ||
        reward < 0
      ) {

        alert(
          "Please enter a valid reward."
        );

        return;

      }

      const button =
        document.getElementById(
          "createJobButton"
        );

      button.disabled = true;
      button.textContent = "Creating...";

      try {

        const response =
          await fetch(
            "/api/admin/jobs",
            {
              method: "POST",

              headers: {
                "Content-Type":
                  "application/json"
              },

              credentials: "same-origin",

              body: JSON.stringify({

                slot: slot,

                title: title,

                category: category,

                url: url,

                description: description,

                reward: reward,

                duration:
                  duration || "Flexible",

                status: status

              })

            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        let data;

        try {
          data = await response.json();
        } catch {
          data = {};
        }

        if (
          !response.ok ||
          !data.success
        ) {

          throw new Error(
            data.message ||
            "Unable to create job."
          );

        }

        alert(
          data.message ||
          `Job ${slot} created successfully.`
        );

        document.getElementById(
          "createJobForm"
        ).reset();

        document.getElementById(
          "jobSlot"
        ).value = "1";

        document.getElementById(
          "jobCategory"
        ).value = "General";

        document.getElementById(
          "jobDuration"
        ).value = "Flexible";

        document.getElementById(
          "jobStatus"
        ).value = "available";

        await loadAdminJobs();

      } catch (error) {

        console.error(
          "Create job error:",
          error
        );

        alert(
          error.message ||
          "Unable to create job."
        );

      } finally {

        button.disabled = false;
        button.textContent = "➕ Create Job";

      }

    }

    /* =========================
       LOAD ADMIN JOBS
    ========================== */

    async function loadAdminJobs() {

      if (!adminAuthenticated) {
        return;
      }

      const table =
        document.getElementById(
          "jobsTable"
        );

      try {

        const response =
          await fetch(
            "/api/admin/jobs?t=" +
            Date.now(),
            {
              method: "GET",
              cache: "no-store",
              credentials: "same-origin"
            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        if (!response.ok) {

          throw new Error(
            "Server returned " +
            response.status
          );

        }

        const data =
          await response.json();

        if (!data.success) {

          throw new Error(
            data.message ||
            "Unable to load jobs."
          );

        }

        const jobs =
          Array.isArray(data.jobs)
            ? data.jobs
            : [];

        jobsCache =
          new Map(
            jobs.map(job => [
              Number(job.id),
              job
            ])
          );

        document.getElementById(
          "totalJobs"
        ).textContent =
          jobs.length;

        if (jobs.length === 0) {

          table.innerHTML = `
            <tr>
              <td colspan="11" class="empty">
                No jobs have been created yet.
              </td>
            </tr>
          `;

          return;

        }

        table.innerHTML =
          jobs.map(job => {

            const status =
              String(
                job.status ||
                "available"
              ).toLowerCase();

            let statusClass = "paused";

            if (
              status === "available" ||
              status === "active"
            ) {

              statusClass = "active";

            } else if (
              status === "completed"
            ) {

              statusClass = "completed";

            }

            let actionButton = "";

            if (
              status === "available" ||
              status === "active"
            ) {

              actionButton = `
                <button
                  type="button"
                  class="job-action pause-job"
                  onclick="changeJobStatus(
                    ${Number(job.id)},
                    'inactive'
                  )"
                >
                  ⏸ Pause
                </button>
              `;

            } else if (
              status === "inactive" ||
              status === "closed"
            ) {

              actionButton = `
                <button
                  type="button"
                  class="job-action activate-job"
                  onclick="changeJobStatus(
                    ${Number(job.id)},
                    'available'
                  )"
                >
                  ▶️ Activate
                </button>
              `;

            } else {

              actionButton = `
                <span
                  style="
                    color:#64748b;
                    font-size:11px;
                    font-weight:600;
                  "
                >
                  Completed
                </span>
              `;

            }

            return `
              <tr>

                <td>
                  <span class="slot">
                    Job ${Number(job.slot || job.job_slot || 0)}
                  </span>
                </td>

                <td>
                  #${Number(job.id)}
                </td>

                <td>
                  <div class="job-title">
                    ${escapeHTML(
                      job.title || "-"
                    )}
                  </div>
                </td>

                <td>
                  <span class="category">
                    ${escapeHTML(
                      job.category || "General"
                    )}
                  </span>
                </td>

                <td>

                  <a
                    class="job-url"
                    href="${escapeHTML(
                      safeUrl(job.url)
                    )}"
                    target="_blank"
                    rel="noopener noreferrer"
                  >
                    ${escapeHTML(
                      job.url || "-"
                    )}
                  </a>

                </td>

                <td>
                  <div class="job-description">
                    ${escapeHTML(
                      job.description || "-"
                    )}
                  </div>
                </td>

                <td>
                  <span class="reward">
                    ${formatNaira(job.reward)}
                  </span>
                </td>

                <td>
                  ${escapeHTML(
                    job.duration || "Flexible"
                  )}
                </td>

                <td>
                  <span
                    class="status ${statusClass}"
                  >
                    ${escapeHTML(status)}
                  </span>
                </td>

                <td>
                  ${escapeHTML(
                    formatDate(job.created_at)
                  )}
                </td>

                <td>
                  ${actionButton}
                </td>

              </tr>
            `;

          }).join("");

      } catch (error) {

        console.error(
          "Load admin jobs error:",
          error
        );

        table.innerHTML = `
          <tr>
            <td colspan="11" class="empty error">
              Unable to connect to the jobs server.
            </td>
          </tr>
        `;

      }

    }

        const response =

            "/api/admin/links?t=" +
            Date.now(),
            {
              method: "GET",
              cache: "no-store",
              credentials: "same-origin"
            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        if (!response.ok) {

          throw new Error(
            "Server returned " +
            response.status
          );

        }

        const data =
          await response.json();

        if (!data.success) {

          throw new Error(
            data.message ||
            "Unable to load links."
          );

        }

        const links =
          Array.isArray(data.links)
            ? data.links
            : [];

        linksCache =
          new Map(
            links.map(link => [
              Number(link.id),
              link
            ])
          );

        document.getElementById(
          "totalLinks"
        ).textContent =
          links.length;

        if (links.length === 0) {

          table.innerHTML = `
            <tr>
              <td colspan="8" class="empty">
                No links have been added yet.
              </td>
            </tr>
          `;

          return;

        }

        table.innerHTML =
          links.map(link => {

            const status =
              String(
                link.status ||
                "active"
              ).toLowerCase();

            const statusClass =
              status === "active"
                ? "link-status-active"
                : "link-status-inactive";

            const url =
              safeUrl(link.url);

            let actionButtons = "";

            if (status === "active") {

              actionButtons += `
                <button
                  type="button"
                  class="link-action pause-link"
                  onclick="changeLinkStatus(
                    ${Number(link.id)},
                    'inactive'
                  )"
                >
                  ⏸ Disable
                </button>
              `;

            } else {

              actionButtons += `
                <button
                  type="button"
                  class="link-action activate-link"
                  onclick="changeLinkStatus(
                    ${Number(link.id)},
                    'active'
                  )"
                >
                  ▶️ Activate
                </button>
              `;

            }

            actionButtons += `
              <button
                type="button"
                class="link-action edit-link"
                onclick="editLink(
                  ${Number(link.id)}
                )"
              >
                ✏️ Edit
              </button>

              <button
                type="button"
                class="link-action delete-link"
                onclick="deleteLink(
                  ${Number(link.id)}
                )"
              >
                🗑️ Delete
              </button>
            `;

            return `
              <tr>

                <td>
                  #${Number(link.id)}
                </td>

                <td>
                  <div class="link-name">
                    ${escapeHTML(
                      link.name ||
                      link.title ||
                      "-"
                    )}
                  </div>
                </td>

                <td>

                  <a
                    class="link-url"
                    href="${escapeHTML(url)}"
                    target="_blank"
                    rel="noopener noreferrer"
                  >
                    ${escapeHTML(
                      link.url || "-"
                    )}
                  </a>

                </td>

                <td>

                  <div class="link-description">
                    ${escapeHTML(
                      link.description || "-"
                    )}
                  </div>

                </td>

                <td>

                  <span class="category">
                    ${escapeHTML(
                      link.category ||
                      "General"
                    )}
                  </span>

                </td>

                <td>

                  <span
                    class="status ${statusClass}"
                  >
                    ${escapeHTML(status)}
                  </span>

                </td>

                <td>
                  ${escapeHTML(
                    formatDate(
                      link.created_at
                    )
                  )}
                </td>

                <td>
                  ${actionButtons}
                </td>

              </tr>
            `;

          }).join("");

      } catch (error) {

        console.error(
          "Load admin links error:",
          error
        );

        table.innerHTML = `
          <tr>
            <td colspan="8" class="empty error">
              Unable to connect to the links server.
            </td>
          </tr>
        `;

      }

    }

       ADD / EDIT LINK
    ========================== */

    async function saveLink(event) {

      event.preventDefault();

      const name =
        document.getElementById(
          "linkName"
        ).value.trim();

      const url =
        document.getElementById(
          "linkUrl"
        ).value.trim();

      const description =
        document.getElementById(
          "linkDescription"
        ).value.trim();

      const category =
        document.getElementById(
          "linkCategory"
        ).value.trim();

      const status =
        document.getElementById(
          "linkStatus"
        ).value;

      if (!name) {

        alert(
          "Please enter a link name."
        );

        return;

      }

      try {

        const parsedUrl =
          new URL(url);

        if (
          parsedUrl.protocol !== "http:" &&
          parsedUrl.protocol !== "https:"
        ) {

          throw new Error();

        }

      } catch {

        alert(
          "Please enter a valid URL beginning with http:// or https://."
        );

        return;

      }

      const button =
        document.getElementById(
          "saveLinkButton"
        );

      button.disabled = true;

      button.textContent =
        editingLinkId
          ? "Updating..."
          : "Adding...";

      try {

        const endpoint =
          editingLinkId
            ? `/api/admin/links/${Number(editingLinkId)}`
            : "/api/admin/links";

        const method =
          editingLinkId
            ? "PUT"
            : "POST";

        const response =
          await fetch(
            endpoint,
            {
              method: method,

              headers: {
                "Content-Type":
                  "application/json"
              },

              credentials: "same-origin",

              body: JSON.stringify({

                name: name,

                url: url,

                description:
                  description,

                category:
                  category || "General",

                status:
                  status

              })

            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        let data;

        try {
          data = await response.json();
        } catch {
          data = {};
        }

        if (
          !response.ok ||
          !data.success
        ) {

          throw new Error(
            data.message ||
            (
              editingLinkId
                ? "Unable to update link."
                : "Unable to add link."
            )
          );

        }

        alert(
          data.message ||
          (
            editingLinkId
              ? "Link updated successfully."
              : "Link added successfully."
          )
        );

        cancelLinkEdit();

        await loadAdminLinks();

      } catch (error) {

        console.error(
          "Save link error:",
          error
        );

        alert(
          error.message ||
          "Unable to save link."
        );

      } finally {

        button.disabled = false;

        button.textContent =
          editingLinkId
            ? "Update Link"
            : "➕ Add Link";

      }

    }

       EDIT LINK
    ========================== */

    function editLink(id) {

      const link =
        linksCache.get(Number(id));

      if (!link) {

        alert(
          "Link information could not be found. Please refresh the links list."
        );

        return;

      }

      editingLinkId = Number(id);

      document.getElementById(
        "linkName"
      ).value =
        link.name ||
        link.title ||
        "";

      document.getElementById(
        "linkUrl"
      ).value =
        link.url || "";

      document.getElementById(
        "linkDescription"
      ).value =
        link.description || "";

      document.getElementById(
        "linkCategory"
      ).value =
        link.category || "General";

      document.getElementById(
        "linkStatus"
      ).value =
        String(
          link.status || "active"
        ).toLowerCase();

      document.getElementById(
        "linkFormTitle"
      ).textContent =
        "✏️ Edit Link";

      document.getElementById(
        "saveLinkButton"
      ).textContent =
        "Update Link";

      document.getElementById(
        "cancelEditButton"
      ).style.display =
        "inline-block";

      document.getElementById(
        "linkFormBox"
      ).classList.add("editing");

      document.getElementById(
        "linkFormBox"
      ).scrollIntoView({
        behavior: "smooth",
        block: "center"
      });

    }

    }

    ========================== */

    async function changeLinkStatus(id, status) {

      const link =
        linksCache.get(Number(id));

      const name =
        link &&
        (link.name || link.title)
          ? (
              link.name ||
              link.title
            )
          : "this link";

      const action =
        status === "inactive"
          ? "disable"
          : "activate";

      const confirmed =
        confirm(
          `Are you sure you want to ${action} ${name}?`
        );

      if (!confirmed) {
        return;
      }

      try {

        const response =
          await fetch(
            `/api/admin/links/${Number(id)}/status`,
            {
              method: "POST",

              headers: {
                "Content-Type":
                  "application/json"
              },

              credentials: "same-origin",

              body: JSON.stringify({
                status: status
              })

            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        let data;

        try {
          data = await response.json();
        } catch {
          data = {};
        }

        if (
          !response.ok ||
          !data.success
        ) {

          throw new Error(
            data.message ||
            "Unable to update link status."
          );

        }

        await loadAdminLinks();

      } catch (error) {

        console.error(
          "Link status error:",
          error
        );

        alert(
          error.message ||
          "Unable to update link."
        );

      }

    }

    /* =========================
       DELETE LINK
    ========================== */

    async function deleteLink(id) {

      const link =
        linksCache.get(Number(id));

      const name =
        link &&
        (link.name || link.title)
          ? (
              link.name ||
              link.title
            )
          : "this link";

      const confirmed =
        confirm(
          `Are you sure you want to permanently delete "${name}"?\n\nThis action cannot be undone.`
        );

      if (!confirmed) {
        return;
      }

      try {

        const response =
          await fetch(
            `/api/admin/links/${Number(id)}`,
            {
              method: "DELETE",
              credentials: "same-origin"
            }
          );

        if (response.status === 401) {

          handleUnauthorized();
          return;

        }

        let data;

        try {
          data = await response.json();
        } catch {
          data = {};
        }

        if (
          !response.ok ||
          !data.success
        ) {

          throw new Error(
            data.message ||
            "Unable to delete link."
          );

        }

        alert(
          data.message ||
          "Link deleted successfully."
        );

        if (
          editingLinkId === Number(id)
        ) {

          cancelLinkEdit();

        }

        await loadAdminLinks();

      } catch (error) {

        console.error(
          "Delete link error:",
          error
        );

        alert(
          error.message ||
          "Unable to delete link."
        );

      }

    }
